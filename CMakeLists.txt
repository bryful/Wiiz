cmake_minimum_required(VERSION 3.24)
project(Wiiz VERSION 0.1.0 LANGUAGES C CXX)
# Keep Android debug builds debuggable while running the existing emulator at full speed.
# Apply before SDL is added so its audio/video adapters are optimized as well.
if(ANDROID)
  add_compile_options("$<$<CONFIG:Debug>:-O2>")
endif()
include(FetchContent)
if(ANDROID)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3
  URL https://github.com/libsdl-org/SDL/releases/download/release-3.2.28/SDL3-3.2.28.zip
  URL_HASH SHA256=24a30069af514a6c6b773bdc8ccca8b321661b251381acc1daeebf8c8f4a109a)
FetchContent_MakeAvailable(SDL3)
if(WIN32)
  include(cmake/PatchSDLSwitch.cmake)
endif()
add_library(binjnes STATIC third_party/binjnes/src/emulator.c third_party/binjnes/src/cartdb.c third_party/binjnes/src/common.c)
target_include_directories(binjnes PUBLIC third_party/binjnes/src)
set_target_properties(binjnes PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_definitions(binjnes PRIVATE BINJNES_MSVC _CRT_SECURE_NO_WARNINGS)
  target_compile_options(binjnes PRIVATE /wd4146 /wd4098)
else()
  if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_definitions(binjnes PRIVATE BINJNES_CLANG)
  else()
    target_compile_definitions(binjnes PRIVATE BINJNES_GCC)
  endif()
endif()
set(WIIZ_SOURCES app/main.cpp app/dungeon.cpp app/core_bridge.c app/session.cpp app/save_json.cpp app/persistence_check.cpp app/spell_map_view.cpp app/rom_identity.cpp app/crt_filter.cpp)
if(ANDROID)
  add_library(wiiz SHARED ${WIIZ_SOURCES})
  set_target_properties(wiiz PROPERTIES OUTPUT_NAME main)
  set_target_properties(binjnes PROPERTIES POSITION_INDEPENDENT_CODE ON)
else()
  add_executable(wiiz ${WIIZ_SOURCES})
endif()
target_compile_features(wiiz PRIVATE cxx_std_20)
target_include_directories(wiiz PRIVATE third_party/picosha2 third_party/nlohmann)
target_link_libraries(wiiz PRIVATE SDL3::SDL3 binjnes)
if(NOT ANDROID)
  add_executable(wiiz_gamepad_check EXCLUDE_FROM_ALL tests/check_gamepad.cpp)
  target_compile_features(wiiz_gamepad_check PRIVATE cxx_std_20)
  target_link_libraries(wiiz_gamepad_check PRIVATE SDL3::SDL3)
endif()
if(WIN32)
  enable_language(RC)
  target_sources(wiiz PRIVATE app/wiiz.rc)
  target_link_libraries(wiiz PRIVATE gdi32)
endif()
set_property(TARGET wiiz PROPERTY C_STANDARD 11)
if(MSVC)
  target_compile_definitions(wiiz PRIVATE BINJNES_MSVC)
  target_compile_options(wiiz PRIVATE /W4 /utf-8)
else()
  target_compile_definitions(wiiz PRIVATE BINJNES_CLANG)
endif()
install(TARGETS wiiz RUNTIME DESTINATION .)
install(DIRECTORY DESTINATION roms)
install(FILES third_party/picosha2/LICENSE DESTINATION licenses RENAME PicoSHA2.txt)
install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME SDL3.txt)
install(DIRECTORY third_party/sdl3/ DESTINATION licenses/SDL3-source)
install(FILES cmake/PatchSDLSwitch.cmake DESTINATION licenses/SDL3-source)
install(FILES third_party/binjnes/LICENSE DESTINATION licenses RENAME binjnes.txt)
install(FILES third_party/binjnes/PALETTE-LICENSE.txt DESTINATION licenses RENAME palgen-persune.txt)
install(FILES third_party/binjnes/PROVENANCE.json third_party/binjnes/wiiz-ram-observer.patch DESTINATION licenses/binjnes-source)
install(FILES docs/turbo.md DESTINATION docs)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt)
install(FILES third_party/wiz-save/LICENSE DESTINATION licenses RENAME WizEdit.txt)
install(FILES third_party/wiz-save/PROVENANCE.json DESTINATION licenses/WizEdit-source)
install(FILES third_party/nlohmann/PROVENANCE.json DESTINATION licenses/nlohmann-json-source)
install(FILES docs/save-data.md DESTINATION docs)
install(FILES docs/spell-map.md DESTINATION docs)
install(FILES docs/rom-identification.md DESTINATION docs)
install(FILES docs/settings.md DESTINATION docs)
install(FILES docs/gamepad.md DESTINATION docs)

install(DIRECTORY third_party/crt-lottes/ DESTINATION licenses/crt-lottes)
install(FILES docs/crt.md DESTINATION docs)

install(FILES docs/difficulty.md DESTINATION docs)

install(FILES docs/creation-bonus.md DESTINATION docs)
install(FILES docs/wiz1-bugfix.md DESTINATION docs)
install(FILES docs/wiz1-patch-notices.txt DESTINATION licenses RENAME Wiz1-patches.txt)

install(FILES docs/roster-order.md DESTINATION docs)

install(FILES third_party/mame-turbofile/LICENSE DESTINATION licenses RENAME MAME-TurboFile.txt)
install(DIRECTORY third_party/mame-turbofile/ DESTINATION licenses/MAME-TurboFile-source)

install(FILES third_party/binjnes/wiiz-expansion-port.patch DESTINATION licenses/binjnes-source)
install(FILES docs/turbofile.md DESTINATION docs)

install(FILES LICENSE DESTINATION .)
